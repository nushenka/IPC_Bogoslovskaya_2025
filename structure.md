/project-root

├── js/
│ ├── config/

│ │   ├── Asset.js            # Базовый класс для активов (Company, Stock, MetalAsset)
│ │   ├── Player.js           # Игрок: капитал, банк, чистая стоимость
│ │   ├── Company.js          # Компания: цена, производство, прибыль, покупка/продажа
│ │   ├── Stock.js            # Акции компаний
│ │   ├── MetalAsset.js       # Инвестиции в металлы
│ │   ├── Demand.js           # Линейный и другие типы спроса
│ │   ├── Cost.js             # Линейные и квадратичные издержки
│ │   ├── Shock.js            # Случайные события и их эффекты
│ │   ├── EconomicEnvironment.js # Экономическая среда: инфляция, ставки, шоки
│ │   └── GovernmentPolicy.js   # Налоги, субсидии, регулирование
│ │
│ ├── game/                   # Логика раундов, симуляция рынка
│ │   ├── roundManager.js     # Управление раундами, переход к следующему
│ │   ├── marketSimulator.js  # Генерация рынка: спрос, цены, конкуренты
│ │   ├── decisionChecker.js  # Проверка решений игрока: объем, прибыль, советы
│ │   └── analyticsManager.js # Сбор всех данных раунда для UI и графиков
│ │
│ └── ui/                     # Отображение и взаимодействие
│     ├── displayCompanies.js  # Функции для списка компаний, кнопки "Купить", "Установить объем"
│     ├── displayGameStatus.js # Капитал, чистая стоимость, отчет игрока
│     ├── renderAnalytics.js   # Отчеты после раунда, текстовые и графические
│     └── charts/              # Модули визуализации графиков
│         ├── demandSupplyChart.js   # График спрос/предложение
│         ├── capitalChart.js        # Динамика капитала игрока
│         └── stockMetalChart.js     # Динамика акций и металлов
│
└── assets/
    ├── images/               # Иконки компаний, кнопок и т.д.
    └── charts/               # Иконки или фоны для графиков
